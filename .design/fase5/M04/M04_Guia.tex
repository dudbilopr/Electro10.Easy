\eSeccion{1}{Ideas clave}
\begin{eClave}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt]
  \item \textbf{De cargas puntuales a fluidos continuos:} cuando la carga está repartida en un cuerpo macroscópico, se trocea en elementos infinitesimales $dq$ mediante la densidad correspondiente: lineal ($dq=\lambda\,d\ell$), superficial ($dq=\sigma\,dA$) o volumétrica ($dq=\rho\,dV$).
  \item \textbf{La suma vectorial se convierte en integral:} cada elemento $dq$ crea un diferencial de campo $d\vect{E}=\frac{k\,dq}{r^2}\,\uv{r}$. La suma de todos los aportes es la integral vectorial $\vect{E}=\int d\vect{E}$. \textbf{Nunca se integran módulos directamente} a menos que todos los vectores apunten en la misma dirección.
  \item \textbf{La simetría elimina componentes antes de integrar:} pares de cargas simétricas cancelan componentes ortogonales al eje de simetría. En la mediatriz de una varilla se cancela $E_x$; sobre el eje de un anillo o disco se cancelan todas las componentes radiales perpendiculares al eje $z$.
  \item \textbf{Los límites asintóticos son el control de calidad:} a distancias muy grandes ($r\gg L$ o $z\gg R$), toda distribución continua finita debe converger exactamente al campo de una carga puntual $kQ/r^2$. Al hacer el radio del disco tender a infinito ($R\to\infty$), el campo se vuelve uniforme: $E=\sigma/(2\epsilon_0)$.
  \item \textbf{Jerarquía de ensamblaje:} un disco se construye apilando anillos concéntricos ($dA=2\pi r'\,dr'$); el plano infinito es un disco de radio infinito.
\end{itemize}
\end{eClave}

\eSeccion{4}{Errores comunes, con su explicación física}
\begin{eError}\textbf{«Sumo algebraicamente los campos integrando $dE$ sin descomponer.»} Grave error. Los diferenciales $d\vect{E}$ apuntan en direcciones distintas en el espacio. Si integras $\int dE$ directamente, obtienes un número mayor al real porque sumas módulos y no vectores. Primero se proyecta: $dE_z = dE\cos\theta$, y solo entonces se integra la componente escalar.\end{eError}
\begin{eError}\textbf{«Uso la fórmula del hilo infinito ($2k\lambda/a$) para una varilla corta de laboratorio.»} La fórmula $E=2k\lambda/a$ solo es válida si $L\gg a$ (al menos $L>10a$). Para una varilla finita de $L=\SI{0,60}{\metre}$ a $a=\SI{0,40}{\metre}$, la fórmula exacta da $\SI{539}{\newton\per\coulomb}$, mientras que el hilo infinito daría $\SI{900}{\newton\per\coulomb}$ ($67\%$ de error).\end{eError}
\begin{eError}\textbf{«En el centro de un anillo cargado el campo es infinito porque $z=0$.»} Al revés: en $z=0$ el campo es estrictamente \textbf{cero}. Para cada $dq$ en el anillo existe otro $dq$ diametralmente opuesto que genera una fuerza igual y de sentido contrario, cancelando el campo por simetría.\end{eError}
\begin{eError}\textbf{«Confundo el campo de una lámina infinita no conductora con el de un conductor.»} Una lámina delgada aislante tiene una sola capa de carga y crea $E=\sigma/(2\epsilon_0)$ a ambos lados. La superficie de un conductor cargado tiene $E=\sigma/\epsilon_0$ porque todo el campo sale hacia el exterior (en el interior $E=0$).\end{eError}

\eSeccion{5}{Cómo estudiar este capítulo}
\begin{eConsejo}Sigue el algoritmo universal en 7 pasos: 1) dibuja la geometría y ubica $dq$; 2) escribe $r$ y el vector unitario; 3) expresa $d\vect{E}$; 4) proyecta con trigonometría ($\cos\theta$ o $\sin\theta$); 5) expresa todas las variables en términos de una sola variable de integración; 6) resuelve la integral definida; 7) comprueba los límites extremos ($z\gg R$ o $a\gg L$). En ContinuousDistLab, interactúa con el deslizador de distancia y observa la transición continua de varilla finita a hilo infinito y de disco a plano.\end{eConsejo}

\eSeccion{6}{Autoevaluación}
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item Una varilla de $L=\SI{0,60}{\metre}$ tiene $Q=+\SI{12}{\nano\coulomb}$. Calcule su campo en la mediatriz a $a=\SI{0,40}{\metre}$.
  \item ¿En qué punto del eje $z$ de un anillo de radio $R$ el campo eléctrico $E_z$ alcanza su valor máximo?
  \item Un disco de radio $R=\SI{0,25}{\metre}$ tiene densidad $\sigma=\SI{6,0}{\micro\coulomb\per\metre\squared}$. Calcule $E$ sobre el eje a $z=\SI{0,30}{\metre}$.
  \item ¿Cuánto vale el campo eléctrico a $\SI{5,0}{\centi\metre}$ y a $\SI{50}{\centi\metre}$ de un plano infinito con $\sigma=\SI{4,0}{\micro\coulomb\per\metre\squared}$?
  \item Si te alejas a $z=\SI{100}{\metre}$ del disco de la pregunta 3, ¿qué fórmula aproximada puedes usar?
\end{enumerate}

{\sffamily\footnotesize\color{e10muted}\textbf{Solucionario.}
1) $E=\frac{kQ}{a\sqrt{a^2+(L/2)^2}}=\frac{(8{,}988\times10^9)(12\times10^{-9})}{0{,}40\sqrt{0{,}40^2+0{,}30^2}}=\SI{539}{\newton\per\coulomb}$.\enspace
2) Derivando $dE_z/dz=0$ se obtiene $z=\pm R/\sqrt{2}$.\enspace
3) $E_z=\frac{\sigma}{2\epsilon_0}[1-z/\sqrt{z^2+R^2}]=\frac{6\times10^{-6}}{2(8{,}854\times10^{-12})}[1-0{,}30/\sqrt{0{,}30^2+0{,}25^2}]=\SI{7,85e4}{\newton\per\coulomb}$.\enspace
4) En ambos puntos vale exactamente lo mismo: $E=\sigma/(2\epsilon_0)=\SI{2,26e5}{\newton\per\coulomb}$ (independiente de la distancia).\enspace
5) La aproximación de carga puntual: $E\approx kQ/z^2=k(\sigma\pi R^2)/z^2$.}

\eSeccion{7}{Conectores de conceptos}
\begin{minipage}[t]{0.48\linewidth}\small
{\sffamily\small\bfseries\color{e10primary}Viene de}
\begin{itemize}[leftmargin=*,itemsep=1pt,topsep=2pt]
  \item \textbf{Cap.~1 · Ley de Coulomb:} interacción puntual entre pares de cargas.
  \item \textbf{Cap.~2 · Campo eléctrico:} $\vect{E}=\sum \vect{E}_i$, la suma vectorial discreta.
  \item \textbf{Cálculo integral:} sumas de Riemann, integrales trigonométricas y sustitución.

\begin{itemize}[leftmargin=*,itemsep=1pt,topsep=2pt]
  \item \textbf{Cap.~5 · Ley de Gauss:} cálculo instantáneo de campos de alta simetría sin integrar $dq$.
  \item \textbf{Cap.~6 · Capacitancia:} campo $E=\sigma/\epsilon_0$ entre placas paralelas de condensadores.
  \item \textbf{Cap.~9 · Biot-Savart:} corrientes continuas divididas en $I\,d\vect{\ell}$.
\end{itemize}

\eSeccion{8}{Glosario}
\begin{multicols}{2}\small\begin{description}[leftmargin=0pt,itemsep=1pt,font=\sffamily\bfseries\color{e10primary}]
  \item[Distribución continua.] Carga repartida densamente en una línea, superficie o volumen macroscópico.
  \item[Densidad lineal $\lambda$.] Carga por unidad de longitud ($dq/d\ell$), unidad $\mathrm{C/m}$.
  \item[Densidad superficial $\sigma$.] Carga por unidad de área ($dq/dA$), unidad $\mathrm{C/m^2}$.
  \item[Densidad volumétrica $\rho$.] Carga por unidad de volumen ($dq/dV$), unidad $\mathrm{C/m^3}$.
  \item[Elemento diferencial $dq$.] Porción infinitesimal de carga tratada como puntual para integrar.
  \item[Límite asintótico.] Comportamiento del campo cuando una dimensión tiende a cero o a infinito ($z\gg R$).
  \item[Cancelación por simetría.] Anulación de componentes vectoriales opuestas debida a la geometría del cuerpo.
  \item[Plano infinito.] Idealización física para regiones cercanas al centro de placas grandes ($z\ll R$).
\end{description}\end{multicols}
