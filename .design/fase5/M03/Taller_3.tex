\ePortada{Capítulo 3}{Taller}{Potencial eléctrico y superficies equipotenciales}{El trabajo por unidad de carga: un escalar que mapea el campo}

\textbf{Qué vas a practicar.} Calcular el potencial eléctrico como la integral de línea del campo y como una suma escalar directa sin vectores; aplicar el principio de superposición a sistemas de cargas discretas y distribuciones simétricas; calcular la energía electrostática total requerida para ensamblar una configuración; aplicar la conservación de la energía mecánica $K_a+U_a=K_b+U_b$ a partículas subatómicas; deducir el vector campo eléctrico a partir de derivadas espaciales del potencial ($\vect{E}=-\vect{\nabla}V$); y trazar superficies equipotenciales ortogonales a las líneas de fuerza.

\begin{eFormula}
\vspace{-6pt}\[ V = \frac{U}{q_0}\qquad \Delta V = -\int_{a}^{b}\vect{E}\cdot d\vect{\ell}\qquad V = k\sum_i \frac{q_i}{r_i}\qquad U = k\sum_{i<j}\frac{q_iq_j}{r_{ij}}\qquad \vect{E}=-\vect{\nabla}V \]\vspace{-10pt}
\end{eFormula}

{\small\color{e10muted}Cómo usar este taller: lee el ejercicio resuelto de cada sección, tapa la solución e intenta resolverlo por ambos caminos (Camino A = formal/integral o conservación; Camino B = directo o trabajo); luego resuelve los retos graduados. Constantes oficiales de Young y Freedman (2009): $k=\SI{8,988e9}{\newton\metre\squared\per\coulomb\squared}$, $\epsilon_0=\SI{8,854e-12}{\coulomb\squared\per\newton\per\metre\squared}$, $e=\SI{1,602e-19}{\coulomb}$, $m_e=\SI{9,109e-31}{\kilogram}$, $m_p=\SI{1,673e-27}{\kilogram}$, $\SI{1}{\electronvolt}=\SI{1,602e-19}{\joule}$.}

\begin{eClave}{\sffamily\small\bfseries\color{e10primary}Antes de empezar, repasa.} El teorema de trabajo y energía cinética $W_{\text{neto}}=\Delta K$; la definición de fuerza conservativa $W=-\Delta U$; la regla de la cadena y derivadas parciales $\partial/\partial x, \partial/\partial y, \partial/\partial z$; y la integral de línea $\int \vect{E}\cdot d\vect{\ell}$. Recuerda que el potencial es un escalar: las cargas positivas suman potencial positivo y las negativas suman potencial negativo, sin proyectar ángulos.\end{eClave}

\eSeccion{3.1}{El potencial de una carga puntual}

\begin{eResuelto}{1 · Potencial por integral de línea y por fórmula directa}
Una carga puntual $q=+\SI{5,0}{\nano\coulomb}$ está fija en el origen. Deduzca analíticamente el potencial $V(r)$ desde el infinito y evalúe su valor a $r=\SI{0,25}{\metre}$.
\begin{center}
\begin{tikzpicture}[scale=1.0,>=Stealth]
  \draw[dashed,e10line] (-0.8,0) -- (5.2,0);
  \fill[e10qpos] (0,0) circle (0.16) node[above=6pt,e10qpos]{\small $q=+5{,}0\ \text{nC}$};
  \fill[e10ink] (2.5,0) circle (0.1) node[above=6pt]{\small $P(r)$};
  \draw[->,very thick,e10accent] (2.5,0) -- (3.7,0) node[above]{\small $\vect{E}$};
  \draw[<-,thick,e10primary] (3.8,-0.4) -- (4.8,-0.4) node[midway,below]{\scriptsize $d\vect{\ell}$ hacia el origen};
  \draw[<->,e10muted] (0,-0.55) -- (2.5,-0.55) node[midway,fill=white,inner sep=2pt]{\small $r=\SI{0,25}{\metre}$};
\end{tikzpicture}
\end{center}
\begin{ePredice}¿El potencial a $\SI{0,25}{\metre}$ será positivo o negativo? ¿Qué pasaría si la carga fuera negativa?\end{ePredice}
\ePaso{1. Camino A — Integral de línea formal desde el infinito}
El campo de la carga puntual es $\vect{E}=(kq/r'^2)\uv{r}$. Al traer una carga de prueba desde el infinito hasta $r$, el desplazamiento es $d\vect{\ell}=dr'\,\uv{r}$, con límites de integración desde $\infty$ hasta $r$:
\[ V(r)=-\int_{\infty}^{r}\vect{E}\cdot d\vect{\ell}=-\int_{\infty}^{r}\frac{kq}{r'^2}\,dr'=-kq\left[-\frac{1}{r'}\right]_{\infty}^{r}=\frac{kq}{r}. \]
\ePaso{2. Camino B — Fórmula directa y evaluación numérica}
Aplicando directamente la definición de potencial de referencia cero en el infinito:
\[ V=\frac{kq}{r}=\frac{(8{,}988\times10^9)(5{,}0\times10^{-9})}{0{,}25}\ \si{\volt}=\SI{179,8}{\volt}\approx\SI{180}{\volt}. \]
\ePaso{3. Verificación física}
El potencial es positivo porque se requiere realizar trabajo positivo contra la fuerza de repulsión eléctrica para traer una carga de prueba positiva desde el infinito. Ambos caminos coinciden con exactitud.
\eResultado{$V(r)=\dfrac{kq}{r}$; evaluado en $r=\SI{0,25}{\metre}$ resulta $V=+\SI{180}{\volt}$.}
\end{eResuelto}

\begin{eReto}{1 \eNivel{nivel básico}}
Calcule el potencial eléctrico a una distancia de $\SI{0,30}{\metre}$ de una carga puntual de $+\SI{2,0}{\nano\coulomb}$.
\end{eReto}

\begin{eReto}{2 \eNivel{nivel básico}}

\end{eReto}

\eSeccion{3.2}{Conservación de la energía mecánica en partículas cargadas}

\begin{eResuelto}{2 · Positrón acelerado por una partícula alfa}
Una partícula alfa ($q_\alpha=+2e$, en reposo) repele a un positrón ($q_p=+e$, masa $m_e=\SI{9,109e-31}{\kilogram}$). Cuando el positrón está a $r_a=\SI{1,00e-10}{\metre}$, se aleja con rapidez $v_a=\SI{3,00e6}{\metre\per\second}$. Halle su rapidez $v_b$ al alcanzar $r_b=\SI{2,00e-10}{\metre}$. \emph{Adaptado de Sears--Zemansky, ejemplo 23.1.}
\begin{center}
\begin{tikzpicture}[scale=1.1,>=Stealth]
  \fill[e10primary] (0,0) circle (0.22) node[below=5pt]{\small Partícula $\alpha$ ($+2e$)};
  \fill[e10qpos] (2.2,0) circle (0.12) node[above=4pt]{\small $r_a$};
  \draw[->,very thick,e10accent] (2.35,0) -- (3.2,0) node[above]{\small $\vect{v}_a$};
  \fill[e10qpos] (4.4,0) circle (0.12) node[above=4pt]{\small $r_b$};
  \draw[->,very thick,e10accent] (4.55,0) -- (5.7,0) node[above]{\small $\vect{v}_b$};
  \draw[dashed,e10line] (0,0) -- (6.0,0);
\end{tikzpicture}
\end{center}
\begin{ePredice}Como la fuerza es de repulsión, ¿la rapidez final $v_b$ será mayor o menor que $v_a$?\end{ePredice}
\ePaso{1. Camino A — Conservación de la energía mecánica $K_a+U_a=K_b+U_b$}
La energía cinética inicial es $K_a=\frac{1}{2}m_ev_a^2=\frac{1}{2}(9{,}109\times10^{-31})(3{,}00\times10^6)^2=\SI{4,10e-18}{\joule}$.
Las energías potenciales eléctricas inicial y final valen:
\[ U_a=\frac{k(2e)(e)}{r_a}=\frac{(8{,}988\times10^9)(2)(1{,}602\times10^{-19})^2}{1{,}00\times10^{-10}}=\SI{4,61e-18}{\joule}, \]
\[ U_b=\frac{k(2e)(e)}{r_b}=\frac{(8{,}988\times10^9)(2)(1{,}602\times10^{-19})^2}{2{,}00\times10^{-10}}=\SI{2,305e-18}{\joule}. \]
Despejando la energía cinética final:
\[ K_b = K_a + U_a - U_b = 4{,}10\times10^{-18} + 4{,}61\times10^{-18} - 2{,}305\times10^{-18} = \SI{6,41e-18}{\joule}. \]
\[ v_b = \sqrt{\frac{2K_b}{m_e}} = \sqrt{\frac{2(6{,}41\times10^{-18})}{9{,}109\times10^{-31}}} = \SI{3,8e6}{\metre\per\second}. \]
\ePaso{2. Camino B — Teorema del trabajo del campo eléctrico}
El campo realiza trabajo positivo sobre el positrón al alejarse: $W_{a\to b}=-\Delta U = U_a-U_b = \SI{2,31e-18}{\joule}$.
Por el teorema trabajo-energía: $W_{a\to b}=K_b-K_a \Rightarrow K_b = K_a + W = \SI{6,41e-18}{\joule}$, lo que reproduce idénticamente $v_b=\SI{3,8e6}{\metre\per\second}$. Si se alejara hasta el infinito ($U_\infty=0$), la rapidez asintótica máxima sería $v_\infty=\sqrt{2(K_a+U_a)/m_e}=\SI{4,4e6}{\metre\per\second}$.
\eResultado{$v_b=\SI{3,8e6}{\metre\per\second}$ (aumentó un $27\%$ debido a la repulsión electrostática).}
\end{eResuelto}

\begin{eReto}{3 \eNivel{nivel básico}}
¿Cuánta energía cinética adquiere una partícula de carga $+1{,}0\ \mu\mathrm{C}$ que parte del reposo y atraviesa una diferencia de potencial de $\SI{500}{\volt}$?
\end{eReto}

\begin{eReto}{4 \eNivel{nivel básico}}

\end{eReto}

\eSeccion{3.3}{Energía potencial de un sistema de cargas y trabajo de ensamble}

\begin{eResuelto}{3 · Ensamblar tres cargas puntuales desde el infinito}
Se tienen $q_1=-e$ en $x=0$, $q_2=+e$ en $x=a$ y se trae $q_3=+e$ desde el infinito hasta $x=2a$, con $a=\SI{0,50}{\nano\metre}$. Halle: (a) el trabajo externo necesario para colocar $q_3$; (b) la energía potencial total del sistema de tres cargas. \emph{Adaptado de Sears--Zemansky, ejemplo 23.2.}
\begin{center}
\begin{tikzpicture}[scale=1.1,>=Stealth]
  \draw[->,e10line] (-0.8,0) -- (5.2,0) node[right]{\small $x$};
  \fill[e10qneg] (0,0) circle (0.15) node[below=5pt]{\small $q_1=-e$};
  \fill[e10qpos] (2.0,0) circle (0.15) node[below=5pt]{\small $q_2=+e$};
  \fill[e10qpos] (4.0,0) circle (0.15) node[below=5pt]{\small $q_3=+e$};
  \draw[<->,e10muted] (0,0.5) -- (2.0,0.5) node[midway,above]{\scriptsize $a$};
  \draw[<->,e10muted] (2.0,0.5) -- (4.0,0.5) node[midway,above]{\scriptsize $a$};
  \draw[<->,e10muted] (0,0.95) -- (4.0,0.95) node[midway,above]{\scriptsize $2a$};
\end{tikzpicture}
\end{center}
\begin{ePredice}¿El trabajo externo será positivo o negativo? (Fíjate en cuál carga está más cerca de $q_3$).\end{ePredice}
\ePaso{1. Camino A — Trabajo como $W_{\text{ext}}=q_3\,V_{1,2}(2a)$}
El potencial creado por $q_1$ y $q_2$ en la posición de llegada $x=2a$ es:
\[ V_{1,2}(2a) = \frac{kq_1}{2a} + \frac{kq_2}{a} = \frac{k(-e)}{2a} + \frac{k(e)}{a} = \frac{ke}{2a}. \]
El trabajo realizado por el agente externo para traer $q_3=+e$ sin acelerar es:
\[ W_{\text{ext}} = q_3\,V_{1,2}(2a) = (e)\left(\frac{ke}{2a}\right) = \frac{ke^2}{2a} = \frac{(8{,}988\times10^9)(1{,}602\times10^{-19})^2}{2(0{,}50\times10^{-9})} = \SI{2,31e-19}{\joule} = \SI{1,44}{\electronvolt}. \]
\ePaso{2. Camino B — Suma de pares sobre todos los enlaces $i<j$}
La energía del estado inicial con dos cargas era $U_{\text{inicial}}=k\frac{q_1q_2}{a}=-\frac{ke^2}{a}$.
Al agregar la tercera carga, la energía potencial total es la suma de los 3 pares:
\[ U_{\text{total}} = k\left[\frac{q_1q_2}{a} + \frac{q_1q_3}{2a} + \frac{q_2q_3}{a}\right] = k\left[\frac{(-e)(e)}{a} + \frac{(-e)(e)}{2a} + \frac{(e)(e)}{a}\right] = -\frac{ke^2}{2a}. \]
El trabajo externo es la diferencia $W_{\text{ext}}=U_{\text{total}}-U_{\text{inicial}}=-\frac{ke^2}{2a}-\left(-\frac{ke^2}{a}\right)=+\frac{ke^2}{2a}=\SI{1,44}{\electronvolt}$.
\eResultado{(a) $W_{\text{ext}}=+\SI{1,44}{\electronvolt}$; (b) $U_{\text{total}}=-\SI{1,44}{\electronvolt}$ (el sistema está ligado electrostáticamente).}
\end{eResuelto}

\begin{eReto}{5 \eNivel{nivel intermedio}}
Tres cargas iguales $+q$ ocupan los vértices de un triángulo equilátero de lado $L$. Calcule la energía electrostática total almacenada en el sistema.
\end{eReto}

\begin{eReto}{6 \eNivel{nivel intermedio}}
Dos cargas de $\pm\SI{6,0}{\nano\coulomb}$ están separadas $\SI{8,0}{\centi\metre}$. ¿Cuánto vale el potencial en el punto medio que las separa? ¿El campo eléctrico allí es cero?
\end{eReto}

\eSeccion{3.4}{Superposición escalar: el dipolo eléctrico}

\begin{eResuelto}{4 · Potencial de un dipolo en tres puntos estratégicos}
Un dipolo está formado por $q_1=+\SI{12}{\nano\coulomb}$ en $y=\SI{5,0}{\centi\metre}$ y $q_2=-\SI{12}{\nano\coulomb}$ en $y=-\SI{5,0}{\centi\metre}$ sobre el eje $y$. Halle el potencial en: (a) punto $a(0,\SI{2,0}{\centi\metre})$; (b) punto $b(0,\SI{14}{\centi\metre})$; (c) punto $c(\SI{13}{\centi\metre},0)$. \emph{Adaptado de Sears--Zemansky, ejemplo 23.4.}
\begin{center}
\begin{tikzpicture}[scale=0.9,>=Stealth]
  \draw[->,e10line] (-0.8,0) -- (4.2,0) node[right]{\small $x$};
  \draw[->,e10line] (0,-2.6) -- (0,3.2) node[above]{\small $y$};
  \fill[e10qpos] (0,1.5) circle (0.15) node[left]{\small $q_1=+12\ \text{nC}$};
  \fill[e10qneg] (0,-1.5) circle (0.15) node[left]{\small $q_2=-12\ \text{nC}$};
  \fill[e10ink] (0,0.6) circle (0.1) node[right]{\small $a(0,2)$};
  \fill[e10ink] (0,2.6) circle (0.1) node[right]{\small $b(0,14)$};
  \fill[e10ink] (3.0,0) circle (0.1) node[below]{\small $c(13,0)$};
\end{tikzpicture}
\end{center}
\ePaso{1. Punto $a$ (entre las dos cargas)}
$r_{1a}=\SI{3,0}{\centi\metre}=\SI{0,030}{\metre}$; $r_{2a}=\SI{7,0}{\centi\metre}=\SI{0,070}{\metre}$.
\[ V_a = k\left[\frac{q_1}{r_{1a}} + \frac{q_2}{r_{2a}}\right] = (8{,}988\times10^9)\left[\frac{12\times10^{-9}}{0{,}030} - \frac{12\times10^{-9}}{0{,}070}\right] = 3595 - 1541 = +\SI{2054}{\volt}. \]
\ePaso{2. Punto $b$ (sobre el eje, exterior)}
$r_{1b}=\SI{9,0}{\centi\metre}=\SI{0,090}{\metre}$; $r_{2b}=\SI{19}{\centi\metre}=\SI{0,19}{\metre}$.
\[ V_b = (8{,}988\times10^9)\left[\frac{12\times10^{-9}}{0{,}090} - \frac{12\times10^{-9}}{0{,}19}\right] = 1198 - 568 = +\SI{630}{\volt}. \]
\ePaso{3. Punto $c$ (en la mediatriz)}
Por simetría, la distancia a ambas cargas es idéntica: $r_{1c}=r_{2c}=\sqrt{13^2+5^2}=\SI{13,9}{\centi\metre}$.
Como $q_1=-q_2$, los dos términos escalares son iguales en módulo y opuestos en signo:
\[ V_c = \frac{kq_1}{r_{1c}} + \frac{kq_2}{r_{2c}} = \frac{k(12\times10^{-9})}{0{,}139} - \frac{k(12\times10^{-9})}{0{,}139} = \SI{0}{\volt}. \]
\eResultado{$V_a=+\SI{2,05}{\kilo\volt}$; $V_b=+\SI{630}{\volt}$; $V_c=\SI{0}{\volt}$ (en la mediatriz el potencial siempre es nulo).}
\end{eResuelto}

\begin{eReto}{7 \eNivel{nivel intermedio}}
Si se coloca una carga $q_0=+\SI{4,0}{\nano\coulomb}$ en los puntos $a$, $b$ y $c$ del ejercicio 4, calcule la energía potencial electrostática $U$ en cada punto.
\end{eReto}

\eSeccion{3.5}{Campo uniforme, aceleradores y electrón-volt}

\begin{eResuelto}{5 · Aceleración de un protón en campo uniforme}
Un protón recorre una distancia $d=\SI{0,50}{\metre}$ en un acelerador lineal con campo uniforme $E=\SI{1,50e7}{\volt\per\metre}$ paralelo a su velocidad. Calcule: (a) la fuerza sobre el protón; (b) el trabajo realizado por el campo; (c) la diferencia de potencial $\Delta V=V_b-V_a$ entre las placas. \emph{Adaptado de Sears--Zemansky, ejemplo 23.3.}
\ePaso{1. Fuerza y trabajo mecánico}
La fuerza eléctrica sobre el protón es constante:
\[ F = qE = (1{,}602\times10^{-19})(1{,}50\times10^7) = \SI{2,40e-12}{\newton}. \]
Como la fuerza y el desplazamiento son paralelos, el trabajo realizado es:
\[ W = F\,d = (2{,}40\times10^{-12})(0{,}50) = \SI{1,20e-12}{\joule} = \frac{\SI{1,20e-12}{\joule}}{\SI{1,602e-19}{\joule\per\electronvolt}} = \SI{7,50e6}{\electronvolt} = \SI{7,50}{\mega\electronvolt}. \]
\ePaso{2. Diferencia de potencial $\Delta V$}
Por definición de potencial eléctrico en un campo uniforme paralelo al eje $x$:
\[ \Delta V = V_b - V_a = -\int_a^b \vect{E}\cdot d\vect{\ell} = -E\,d = -(1{,}50\times10^7)(0{,}50) = -\SI{7,50e6}{\volt} = -\SI{7,50}{\mega\volt}. \]
Nótese que la carga positiva se mueve en el sentido del campo, cayendo a un potencial menor ($\Delta V < 0$).
\eResultado{(a) $F=\SI{2,40e-12}{\newton}$; (b) $W=\SI{7,50}{\mega\electronvolt}$; (c) $\Delta V=-\SI{7,50}{\mega\volt}$.}
\end{eResuelto}

\begin{eReto}{8 \eNivel{nivel intermedio}}

\end{eReto}

\eSeccion{3.6}{Potencial de distribuciones simétricas y el gradiente}

\begin{eResuelto}{6 · Anillo cargado y deducción del campo sobre el eje}
Un anillo de radio $R$ tiene carga total $Q$ distribuida uniformemente. (a) Halle el potencial $V(z)$ en un punto sobre el eje a distancia $z$ del plano del anillo. (b) A partir de $V(z)$, obtenga el campo eléctrico $E_z$ mediante el gradiente $\vect{E}=-\vect{\nabla}V$.
\begin{center}
\begin{tikzpicture}[scale=1.0,>=Stealth]
  \draw[e10primary,thick] (0,0) ellipse (0.45 and 1.2);
  \draw[dashed,e10line] (0,0) -- (4.2,0) node[right]{\small $z$};
  \fill[e10qpos] (0,1.2) circle (0.1) node[above]{\small $dq$};
  \fill[e10ink] (2.8,0) circle (0.1) node[below]{\small $P(0,0,z)$};
  \draw[->,e10accent,thick] (0,1.2) -- (2.8,0) node[midway,above]{\scriptsize $r=\sqrt{z^2+R^2}$};
  \draw[->,very thick,e10accentstrong] (2.8,0) -- (3.9,0) node[above]{\small $E_z$};
\end{tikzpicture}
\end{center}
\ePaso{1. Cálculo del potencial escalar}
Cada elemento diferencial $dq$ del anillo se encuentra exactamente a la misma distancia de $P$: $r=\sqrt{z^2+R^2}$. Como $r$ es constante durante la integración sobre todo el anillo:
\[ V(z) = \int \frac{k\,dq}{r} = \frac{k}{\sqrt{z^2+R^2}}\int dq = \frac{kQ}{\sqrt{z^2+R^2}}. \]
\ePaso{2. Obtención de $\vect{E}$ mediante la derivada espacial}
Por simetría, el campo no tiene componentes en $x$ ni en $y$. La componente axial se obtiene mediante el gradiente:
\[ E_z = -\frac{\partial V}{\partial z} = -\frac{d}{dz}\left[kQ(z^2+R^2)^{-1/2}\right] = -kQ\left(-\frac{1}{2}\right)(z^2+R^2)^{-3/2}(2z) = \frac{kQz}{(z^2+R^2)^{3/2}}. \]
\ePaso{3. Verificación}
El resultado coincide exactamente con la deducción por componentes vectoriales del capítulo 2, pero requirió muchas menos líneas de cálculo al derivar una función escalar en lugar de integrar vectores trigonométricos.
\eResultado{$V(z)=\dfrac{kQ}{\sqrt{z^2+R^2}}$;\quad $E_z=-\dfrac{dV}{dz}=\dfrac{kQz}{(z^2+R^2)^{3/2}}$.}
\end{eResuelto}

\begin{eReto}{10 \eNivel{nivel intermedio}}

\end{eReto}

\eSeccion{T}{Preguntas conceptuales de teoría}

\begin{eTeoria}
\textbf{¿Por qué es mucho más sencillo calcular el potencial $V$ que el campo eléctrico $\vect{E}$?}\par
\emph{Respuesta.} Porque el potencial es un escalar: no tiene dirección ni sentido, y se suma con álgebra común y corriente ($\sum V_i$), sin tener que calcular componentes rectangulares ni descomponer cosenos o senos.
\end{eTeoria}

\begin{eTeoria}
\textbf{¿Por qué las líneas de campo eléctrico siempre son perpendiculares a las superficies equipotenciales?}\par
\emph{Respuesta.} Si el campo tuviera una componente paralela a la superficie equipotencial, al mover una carga sobre ella se realizaría trabajo ($dW=\vect{E}\cdot d\vect{\ell}\neq0$), lo que violaría la definición de que la diferencia de potencial es nula ($\Delta V=0\Rightarrow W=0$). Por lo tanto, el campo eléctrico debe ser rigurosamente ortogonal a la superficie en todo punto.
\end{eTeoria}

\begin{eTeoria}
\textbf{¿Qué representa físicamente el signo negativo en la relación $\vect{E}=-\vect{\nabla}V$?}\par
\emph{Respuesta.} El operador gradiente $\vect{\nabla}V$ apunta en la dirección de máximo incremento del potencial. El signo menos garantiza que el vector campo eléctrico apunte en el sentido opuesto: hacia donde el potencial disminuye con mayor rapidez (cuesta abajo en el paisaje electrostático).
\end{eTeoria}

\textbf{1. Cálculo automatizado de voltajes y campos.} Este script evalúa los problemas del taller y verifica la conservación de energía:
\begin{lstlisting}[style=e10python]
import numpy as np

K = 8.988e9
E = 1.602e-19
ME = 9.109e-31

# P1: Potencial de carga puntual
q = 5.0e-9; r = 0.25
V1 = K * q / r
print(f"P1: V = {V1:.1f} V (Esperado: 180 V)")

# P4: Dipolo en puntos a, b, c
q_dip = 12.0e-9
Va = K * (q_dip/0.03 - q_dip/0.07)
Vb = K * (q_dip/0.09 - q_dip/0.19)
Vc = K * (q_dip/0.139 - q_dip/0.139)
print(f"P4: Va = {Va:.0f} V, Vb = {Vb:.0f} V, Vc = {Vc:.0f} V")

# P6: Anillo sobre el eje
R_anillo = 0.06; z_anillo = 0.08; Q_anillo = 8.0e-9
V_anillo = K * Q_anillo / np.sqrt(z_anillo**2 + R_anillo**2)
print(f"P6: V_anillo = {V_anillo:.1f} V")
\end{lstlisting}

\textbf{2. Trazador de curvas de nivel equipotenciales.} Generación de mapa de contornos:
\begin{lstlisting}[style=e10python]
import numpy as np

# Malla 2D de potencial para un dipolo
x = np.linspace(-0.2, 0.2, 200)
y = np.linspace(-0.2, 0.2, 200)
X, Y = np.meshgrid(x, y)

# Posiciones de cargas +/- q en el eje y
d = 0.05
r_pos = np.sqrt(X**2 + (Y - d)**2)
r_neg = np.sqrt(X**2 + (Y + d)**2)
V = 8.988e9 * 1e-9 * (1.0/r_pos - 1.0/r_neg)
print(f"Mapa equipotencial generado con exito. Rango V: [{V.min():.1e}, {V.max():.1e}]")
\end{lstlisting}

\eSeccion{C}{Conecta y consulta}
\begin{eConsejo}\textbf{Lo que sigue.} En el capítulo 4, la suma discreta $V=k\sum q_i/r_i$ se generaliza al cálculo integral sobre distribuciones continuas de carga: $V=k\int \frac{dq}{r}$. Calcular el escalar $V$ primero y obtener el vector $\vect{E}$ derivando ($\vect{E}=-\vect{\nabla}V$) es el método más rápido y elegante de la física teórica.\end{eConsejo}
