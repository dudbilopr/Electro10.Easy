\eSeccion{1}{Ideas clave}
\begin{eClave}
\begin{itemize}[leftmargin=*,itemsep=2pt,topsep=0pt]
  \item \textbf{El potencial es energía por unidad de carga:} $V=U/q_0$, en volt ($1\ \mathrm{V}=1\ \mathrm{J/C}$). Es un \textbf{escalar}: no tiene componentes vectoriales ni requiere descomposición trigonométrica para sumarse.
  \item \textbf{El voltaje es la altura de un mapa topográfico:} el campo eléctrico $\vect{E}$ apunta cuesta abajo, en la dirección de máximo descenso del potencial. Una carga positiva acelera espontáneamente hacia menor $V$; una negativa, hacia mayor $V$.
  \item \textbf{Superficies equipotenciales = curvas de nivel:} el campo eléctrico siempre es \textbf{perpendicular} ($\mathbf{90^\circ}$) a las superficies equipotenciales. Mover una carga sobre una misma superficie no requiere trabajo: $W_{a\to b}=0$ si $V_a=V_b$.
  \item \textbf{$V=0$ no implica $\vect{E}=\vect{0}$:} en el punto medio de un dipolo eléctrico el potencial se cancela ($V=0$), pero el campo eléctrico es máximo y distinto de cero. Lo que determina el campo es la derivada (pendiente), no el valor absoluto de $V$.
  \item \textbf{Conductores en equilibrio:} todo el volumen y la superficie de un conductor cargado en equilibrio electrostático se encuentran al \textbf{mismo potencial} ($V=\text{cte}$), por lo que $\vect{E}=\vect{0}$ en su interior.
  \item \textbf{El electrón-volt es unidad de energía, no de potencial:} $1\ \mathrm{eV}=e\cdot(1\ \mathrm{V})=\SI{1,602e-19}{\joule}$.
\end{itemize}
\end{eClave}

\eSeccion{4}{Errores comunes, con su explicación física}
\begin{eError}\textbf{«Sumo los potenciales como vectores con senos y cosenos.»} No: el potencial eléctrico es un campo escalar, no vectorial. Se suma con signo algebraico ordinario: si tienes dos cargas de $+4\ \mathrm{nC}$ y $-4\ \mathrm{nC}$ a la misma distancia, $V=V_1+V_2=0\ \mathrm{V}$, sin descomponer ángulos.\end{eError}
\begin{eError}\textbf{«Si el potencial en un punto es cero, el campo eléctrico también debe ser cero.»} Falso. El campo eléctrico es la derivada espacial (la pendiente) del potencial: $\vect{E}=-\vec\nabla V$. En el plano medio de un dipolo $V=0\ \mathrm{V}$, pero la pendiente es pronunciada y el campo es intenso.\end{eError}
\begin{eError}\textbf{«El trabajo para mover una carga depende de la trayectoria elegida.»} Falso. La fuerza electrostática es estrictamente conservativa: $\oint \vect{E}\cdot d\vect{\ell}=0$. El trabajo $W_{a\to b}=-q\Delta V$ depende exclusivamente de los puntos inicial y final, nunca de la curva recorrida.\end{eError}
\begin{eError}\textbf{«El electrón-volt es un voltaje.»} El electrón-volt ($\mathrm{eV}$) es una unidad de energía ($1\ \mathrm{eV}=\SI{1,602e-19}{\joule}$), equivalente a la energía cinética que gana un electrón al atravesar una diferencia de potencial de $\SI{1}{\volt}$.\end{eError}

\eSeccion{5}{Cómo estudiar este capítulo}
\begin{eConsejo}Domina la analogía del mapa de curvas de nivel: el potencial es la cota de altitud y el campo eléctrico es la flecha de máxima pendiente cuesta abajo. En EquipotentialLab, visualiza cómo al colocar cargas puntuales se forman pozos (cargas negativas) y colinas (cargas positivas). Resuelve los problemas primero calculando el escalar $V$ y sólo después el vector $\vect{E}$ mediante derivadas parciales; este orden siempre simplifica el cálculo analítico.\end{eConsejo}

\eSeccion{6}{Autoevaluación}
\begin{enumerate}[leftmargin=*,itemsep=3pt]
  \item ¿Cuánto vale el potencial a $\SI{0,25}{\metre}$ de una carga puntual de $+\SI{5,0}{\nano\coulomb}$?
  \item Si un protón se acelera desde el reposo a través de $\Delta V=-\SI{500}{\volt}$, ¿cuál es su energía cinética final en $\mathrm{eV}$ y en joules?
  \item Dos cargas de $+q$ y $-q$ están separadas una distancia $2a$. ¿Cuánto vale el potencial en cualquier punto de su mediatriz?
  \item Si el potencial en una región está dado por $V(x,y)=4x^2y - 2z$, halle las componentes del vector $\vect{E}$.
  \item ¿Cuánto trabajo se requiere para mover una carga de $\SI{2,0}{\micro\coulomb}$ a lo largo de una superficie equipotencial de $\SI{100}{\volt}$ durante $\SI{5}{\metre}$?
\end{enumerate}

\eSeccion{8}{Glosario}
\begin{multicols}{2}\small\begin{description}[leftmargin=0pt,itemsep=1pt,font=\sffamily\bfseries\color{e10primary}]
  \item[Potencial eléctrico $V$.] Energía potencial electrostática por unidad de carga; escalar medido en volts ($1\ \mathrm{V}=1\ \mathrm{J/C}$).
  \item[Voltaje $\Delta V$.] Diferencia de potencial $V_b-V_a=-\int_a^b\vect{E}\cdot d\vect{\ell}$ entre dos puntos.
  \item[Superficie equipotencial.] Lugar geométrico de los puntos que comparten el mismo valor de $V$; $\vect{E}\perp$ a ella.
  \item[Gradiente de potencial.] Vector $\vect{\nabla}V=(\frac{\partial V}{\partial x},\frac{\partial V}{\partial y},\frac{\partial V}{\partial z})$ que apunta hacia el máximo aumento de $V$; $\vect{E}=-\vect{\nabla}V$.
  \item[Electrón-volt ($\mathrm{eV}$).] Unidad de energía correspondiente al trabajo realizado al mover $e$ a través de $\SI{1}{\volt}$; $\SI{1,602e-19}{\joule}$.
  \item[Conductor equipotencial.] Todo cuerpo conductor en equilibrio electrostático mantiene el mismo potencial en su superficie y volumen.
  \item[Rigidez dieléctrica.] Campo eléctrico máximo que soporta un aislante antes de ionizarse y conducir ($3\times10^6\ \mathrm{V/m}$ en aire).
  \item[Energía de ensamble.] Trabajo externo necesario para reunir una configuración de cargas desde el infinito: $U=k\sum_{i<j}\frac{q_iq_j}{r_{ij}}$.
\end{description}\end{multicols}
